cmake_minimum_required(VERSION 3.10)
project(ddnet_demo C)
if(MSVC)
  add_definitions(-D_CRT_SECURE_NO_WARNINGS)
endif()
add_library(ddnet_demo INTERFACE)
target_include_directories(ddnet_demo INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})

# Demo reconstruction needs the map loader for its internal physics (internal/).
# A parent project may already provide ddnet_map_loader; reuse it if so.
if(NOT TARGET ddnet_map_loader)
  add_subdirectory(ddnet_maploader)
endif()
option(EXAMPLES "Build the example executables" OFF)
if(EXAMPLES)
  add_executable(example_write example_write.c)
  if(UNIX)
    target_link_libraries(example_write PRIVATE m)
  endif()
  add_executable(example_read example_read.c)
  add_executable(test_ddnet_demo test_ddnet_demo.c)
  # Reconstruction tests against real demos; they use the internal physics.
  foreach(test_name test_reckoning test_coverage test_state test_cache)
    add_executable(${test_name} ${test_name}.c)
    target_link_libraries(${test_name} PRIVATE ddnet_demo ddnet_map_loader)
    if(UNIX)
      target_link_libraries(${test_name} PRIVATE m)
    endif()
  endforeach()
endif()
